const SHEET_ID = "1wzhN9IlL6e_SsrRJh8g_Z1-9vq-pUBQgMNDcUYrHUd8";
const SHEET_NAME = "Applications";

function getSheet() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) sheet = ss.insertSheet(SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Application No", "Name", "Mobile", "WhatsApp", "Apply For", "Date & Time", "Status"]);
  } else if (sheet.getLastColumn() < 7) {
    sheet.getRange(1, 7).setValue("Status");
    const rows = sheet.getLastRow();
    if (rows > 1) sheet.getRange(2, 7, rows - 1, 1).setValue("New");
  }
  return sheet;
}

function doPost(e) {
  try {
    const data = e.parameter || {};
    const sheet = getSheet();

    if (data.action === "delete") {
      const applicationNo = String(data.applicationNo || "").trim();
      const values = sheet.getDataRange().getValues();
      for (let i = values.length - 1; i >= 1; i--) {
        if (String(values[i][0]) === applicationNo) {
          sheet.deleteRow(i + 1);
          return json({success:true});
        }
      }
      throw new Error("Application not found.");
    }

    if (data.action === "status") {
      const applicationNo = String(data.applicationNo || "").trim();
      const status = String(data.status || "New").trim();
      if (["New","Processing","Completed"].indexOf(status) === -1) throw new Error("Invalid status.");
      const values = sheet.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][0]) === applicationNo) {
          sheet.getRange(i + 1, 7).setValue(status);
          return json({success:true});
        }
      }
      throw new Error("Application not found.");
    }

    sheet.appendRow([
      data.applicationNo || "", data.name || "", data.mobile || "", data.whatsapp || "",
      data.reason || "", data.createdAt || new Date().toLocaleString("en-IN"), "New"
    ]);
    return json({success:true});
  } catch(error) {
    return json({success:false,error:String(error)});
  }
}

function doGet(e) {
  try {
    const sheet = getSheet();
    const values = sheet.getDataRange().getValues();
    const applications = values.slice(1).map(row => ({
      applicationNo: row[0], name: row[1], mobile: row[2], whatsapp: row[3],
      reason: row[4], createdAt: row[5], status: row[6] || "New"
    }));
    return json({success:true, applications}, e && e.parameter ? e.parameter.callback : "");
  } catch(error) {
    return json({success:false,error:String(error),applications:[]}, e && e.parameter ? e.parameter.callback : "");
  }
}

function json(data, callback) {
  const output = JSON.stringify(data);
  if (callback && /^[A-Za-z_$][0-9A-Za-z_$]*$/.test(callback)) {
    return ContentService.createTextOutput(callback + "(" + output + ");").setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(output).setMimeType(ContentService.MimeType.JSON);
}
